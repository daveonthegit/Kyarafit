                        objectPosition:
                          build.imageFocalX != null && build.imageFocalY != null
                            ? `${build.imageFocalX * 100}% ${build.imageFocalY * 100}%`

                      }}
                    />
                    <div
                      className="absolute inset-0 bg-gradient-to-t from-black/85 via-black/35 to-black/10"
                      aria-hidden
                    />
                  </div>
                  <div className="relative z-10 flex flex-col justify-end min-h-[min(52vw,380px)] sm:min-h-[400px] p-6 sm:p-10">
                    <p className="text-white/75 text-[10px] sm:text-[11px] uppercase tracking-[0.2em] mb-3">
                      {build.status}
                    </p>
                    <h1 className="font-serif text-4xl sm:text-5xl lg:text-6xl font-bold text-white leading-[1.05] tracking-tight">
                      {build.name}
                    </h1>
                    {build.character && (
                      <p className="text-white/85 mt-3 text-base sm:text-lg">{build.character}</p>
                    )}
                  </div>
                </section>

              <div className="flex flex-col gap-10 xl:grid xl:grid-cols-12 xl:gap-x-8 xl:gap-y-10 xl:items-start">
                <div className="xl:col-span-1 flex xl:justify-center xl:pt-2">
                  <EditorialVerticalProgressRail progress={completionPercent} />
                </div>

                <div className="xl:col-span-3 space-y-6 min-w-0 xl:sticky xl:top-24">
                  {!(build.imageStorageId || build.imageUrl) && (
                    <div>
                      <p className="text-[10px] uppercase tracking-wide text-kyar-textTertiary mb-2">
                        {build.status}
                      </p>
                      <h1 className="font-serif text-3xl lg:text-4xl font-semibold tracking-tight mb-2">
                        {build.name}
                      </h1>
                      {build.character && (
                        <p className="text-sm text-kyar-textTertiary">Character: {build.character}</p>
                      )}
                    </div>
                  )}
                  <div className="flex flex-col items-start gap-2">
                    <EditorialProgressDonut showFlankLabels progress={completionPercent} />
                    <p className="text-xs text-kyar-textTertiary leading-snug">
                      {tasksChecked} of {tasksTotal} tasks complete
                    </p>
                  </div>
                  {build.budgetCents != null && (
                    <div>
                      <span className="text-[10px] uppercase tracking-[0.2em] font-medium text-kyar-textTertiary block mb-2">
                        Budget summary
                      </span>
                      <p className="text-sm text-kyar-text">
                        {formatCents(totalCostCents)} / {formatCents(build.budgetCents)}
                      </p>

                          className="h-full bg-kyar-text transition-all"

                          }}

                      </div>
                      {totalCostCents > (build.budgetCents || 0) && (
                        <p className="text-xs text-red-600 mt-1">
                          Over budget by {formatCents(totalCostCents - (build.budgetCents || 0))}
                        </p>
                      )}
                    </div>

                  {build.targetDate && daysRemaining !== null && (
                    <div>
                      <span className="text-[10px] uppercase tracking-[0.2em] font-medium text-kyar-textTertiary block mb-2">
                        Deadline

                      <div className="flex items-center gap-2">
                        <span className="material-symbols-outlined text-lg">event</span>
                        <div>
                          <p className="text-sm font-medium">
                            {new Date(build.targetDate).toLocaleDateString(undefined, {
                              year: "numeric",
                              month: "long",
                              day: "numeric",
                            })}
                          </p>
                          <p
                            className={`text-xs ${
                              daysRemaining < 0
                                ? "text-red-600"
                                : daysRemaining <= 7
                                  ? "text-orange-600"
                                  : "text-kyar-textTertiary"
                            }`}
                          >
                            {daysRemaining < 0
                              ? `${Math.abs(daysRemaining)} days overdue`
                              : daysRemaining === 0

                                : `${daysRemaining} days remaining`}
                          </p>

                        onClick={handleCopyShareLink}

                                  {(c.name ?? c.email ?? "?").charAt(0).toUpperCase()}
                                </span>

                              <span className="text-[11px] text-kyar-textTertiary capitalize shrink-0">
                                {c.role}
                              </span>
                              <button

                                onClick={() =>
                                  removeCollaborator({
                                    buildId: id,
                                    ownerId: userId,
                                    userId: c.userId,

                                }
                                className="text-[11px] text-red-600 hover:underline shrink-0"
                              >
                                Remove
                              </button>
                            </li>

                      <div className="flex gap-2 flex-wrap items-center">

                          placeholder="Email to share with"
                          className="flex-1 min-w-[140px] border border-kyar-cardBorder rounded px-2 py-1.5 text-sm"
                          disabled={collabPending}
                        />
                        <select
                          value={collabRole}
                          onChange={(e) => setCollabRole(e.target.value as "viewer" | "editor")}
                          className="border border-kyar-cardBorder rounded px-2 py-1.5 text-sm"
                        >
                          <option value="viewer">Viewer</option>
                          <option value="editor">Editor</option>
                        </select>
                        <button
                          type="button"
                          onClick={async () => {
                            if (!id || !userId || !collabEmail.trim() || collabPending) return;
                            setCollabPending(true);
                            setCollabError(null);
                            try {
                              await addCollaborator({
                                buildId: id,

                                role: collabRole,

                              setCollabPending(false);

                          className="text-[11px] uppercase tracking-widest font-medium text-kyar-accent hover:underline disabled:opacity-50"

                  <section ref={tasksSectionRef} id="build-tasks">
                    <h2 className="font-serif text-2xl text-kyar-text border-b border-kyar-borderSubtle pb-2 mb-4">
                      Tasks &amp; Timeline
                    </h2>
                    <p className="text-xs text-kyar-textTertiary mb-4">
                      Drag tasks onto closet items in the Visual Board (Closet tab), or use the link
                      button on a task to assign.
                    </p>
                    <TaskChecklist
                      buildId={id}
                      tasks={tasks}
                      linkedItems={linkedItems}
                      enableDragDrop
                    />
                  </section>
                </div>

                <div ref={itemsSectionRef} id="build-items" className="xl:col-span-3 min-w-0">
                  {userId ? (
                    <BuildVisualBoard
                      buildId={id}
                      userId={userId}
                      linkedItems={linkedItems}
                      linkItemsHref={`/build-detail/link-items?id=${id}`}
                      renderClosetCard={(item) => (
                        <DroppableClosetItem item={item} justDroppedRef={justDroppedRef}>
                          <ClosetCarouselCardContent item={item} formatCents={formatCents} />
                        </DroppableClosetItem>
                      )}
                    />
                  ) : (
                    <p className="text-sm text-kyar-textTertiary">Sign in to manage the visual board.</p>
                  )}
                </div>
              </div>
